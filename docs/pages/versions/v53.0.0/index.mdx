// App.js // Full Expo React Native single-file MVP for "CarpenterConnect" (Hindi/English + easy UI + demo OTP + AsyncStorage) // Ready to paste into an Expo project (expo init) as App.js

import React, { useEffect, useState } from 'react'; import { SafeAreaView, View, Text, TextInput, TouchableOpacity, FlatList, Alert, StyleSheet, Linking, Platform } from 'react-native'; import AsyncStorage from '@react-native-async-storage/async-storage'; import * as Speech from 'expo-speech';

// ---------- TRANSLATIONS (simple inline i18n) ---------- const STRINGS = { hi: { appName: 'कारपेंटर कनेक्ट', welcome: 'स्वागत है', chooseRole: 'अपना रोल चुनें', customer: 'ग्राहक', carpenter: 'बढ़ई', startReg: 'रजिस्ट्रेशन शुरू करें', seeList: 'लिस्ट देखें', enterPhone: 'फोन नंबर डालें', nameOptional: 'नाम (ऐच्छिक)', sendOtp: 'OTP भेजें (डेमो)', enterOtp: 'OTP दर्ज करें', verify: 'सत्यापित करें', call: 'कॉल करें', noCarpenters: 'कोई बढ़ई उपलब्ध नहीं', back: 'पीछे जाएँ', langSwitch: 'भाषा बदलें', demoOtpMsg: 'यह एक डेमो OTP है: ', registered: 'रजिस्ट्रेशन पूरा', wrongOtp: 'गलत OTP', invalidPhone: 'सही फोन नंबर डालें', help: 'मदद के लिए कॉल करें' }, en: { appName: 'Carpenter Connect', welcome: 'Welcome', chooseRole: 'Choose your role', customer: 'Customer', carpenter: 'Carpenter', startReg: 'Start Registration', seeList: 'See List', enterPhone: 'Enter phone number', nameOptional: 'Name (optional)', sendOtp: 'Send OTP (demo)', enterOtp: 'Enter OTP', verify: 'Verify', call: 'Call', noCarpenters: 'No carpenters available', back: 'Back', langSwitch: 'Switch Language', demoOtpMsg: 'This is a demo OTP: ', registered: 'Registration complete', wrongOtp: 'Wrong OTP', invalidPhone: 'Please enter a valid phone', help: 'Call for help' } };

// ---------- Helper functions ---------- const generateOTP = () => Math.floor(100000 + Math.random() * 900000).toString();

const STORAGE_KEY = '@carpenter_connect_carpenters_v1';

// ---------- App Component ---------- export default function App() { const [lang, setLang] = useState('hi'); // default Hindi const strings = STRINGS[lang];

const [screen, setScreen] = useState('home'); // home | register | verify | list const [role, setRole] = useState('customer'); const [name, setName] = useState(''); const [phone, setPhone] = useState(''); const [otpSent, setOtpSent] = useState(null); const [otpInput, setOtpInput] = useState(''); const [carpenters, setCarpenters] = useState([]);

useEffect(() => { loadCarpenters(); }, []);

useEffect(() => { // speak welcome on home if (screen === 'home') { speak(${strings.welcome}. ${strings.chooseRole}); } }, [screen, lang]);

const speak = (text) => { try { Speech.speak(text, { language: lang === 'hi' ? 'hi-IN' : 'en-US' }); } catch (e) { } };

const loadCarpenters = async () => { try { const json = await AsyncStorage.getItem(STORAGE_KEY); if (json) setCarpenters(JSON.parse(json)); } catch (e) { console.log('load err', e); } };

const saveCarpenters = async (arr) => { try { await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(arr)); } catch (e) { console.log('save err', e); } };

const startRegister = () => { // simple validation for phone const p = phone.replace(/\s+/g, ''); if (!/^[6-9]\d{9}$/.test(p)) { Alert.alert(strings.invalidPhone); speak(strings.invalidPhone); return; } const otp = generateOTP(); setOtpSent(otp); // Demo: show otp in alert AND speak it Alert.alert(strings.demoOtpMsg + otp); speak(strings.demoOtpMsg + otp); setScreen('verify'); };

const verifyOtpAndSave = async () => { if (otpInput.trim() === otpSent) { if (role === 'carpenter') { const newCarp = { id: Date.now().toString(), name: name || (lang === 'hi' ? 'अज्ञात' : 'Unknown'), phone: phone, skills: lang === 'hi' ? 'सामान्य' : 'General', city: lang === 'hi' ? 'नहीं बताया' : 'Not set' }; const updated = [newCarp, ...carpenters]; setCarpenters(updated); await saveCarpenters(updated); Alert.alert(strings.registered); speak(strings.registered); } else { Alert.alert(strings.registered); speak(strings.registered); } // reset and go to list so customer can call setName(''); setPhone(''); setOtpInput(''); setOtpSent(null); setScreen('list'); } else { Alert.alert(strings.wrongOtp); speak(strings.wrongOtp); } };

const callNumber = (num) => { const url = tel:${num}; Linking.canOpenURL(url).then((supported) => { if (supported) Linking.openURL(url); else Alert.alert('Error', 'Cannot open phone app'); }); };

const renderHome = () => ( <View style={styles.card}> <Text style={styles.title}>{strings.appName}</Text> <Text style={styles.subtitle}>{strings.chooseRole}</Text>

<View style={{flexDirection: 'row', justifyContent: 'space-between', marginVertical: 12}}>
    <TouchableOpacity onPress={() => { setRole('customer'); speak(strings.customer); }} style={[styles.roleBtn, role === 'customer' && styles.roleBtnActive]}>
      <Text style={styles.roleText}>👤 {strings.customer}</Text>
    </TouchableOpacity>

    <TouchableOpacity onPress={() => { setRole('carpenter'); speak(strings.carpenter); }} style={[styles.roleBtn, role === 'carpenter' && styles.roleBtnActive]}>
      <Text style={styles.roleText}>🔨 {strings.carpenter}</Text>
    </TouchableOpacity>
  </View>

  <TouchableOpacity onPress={() => setScreen('register')} style={styles.primaryBtn}><Text style={styles.primaryBtnText}>{strings.startReg}</Text></TouchableOpacity>
  <TouchableOpacity onPress={() => setScreen('list')} style={styles.ghostBtn}><Text>{strings.seeList}</Text></TouchableOpacity>

  <View style={{flexDirection:'row', justifyContent:'space-between', marginTop:20}}>
    <TouchableOpacity onPress={() => { setLang(prev => prev === 'hi' ? 'en' : 'hi'); speak(lang === 'hi' ? 'Language switched to English' : 'भाषा बदल दी गई'); }} style={styles.smallBtn}>
      <Text>{strings.langSwitch}</Text>
    </TouchableOpacity>

    <TouchableOpacity onPress={() => { Alert.alert(strings.help, '9876543210'); }} style={styles.smallBtn}><Text>{strings.help}</Text></TouchableOpacity>
  </View>
</View>

);

const renderRegister = () => ( <View style={styles.card}> <Text style={styles.h2}>{role === 'carpenter' ? strings.carpenter : strings.customer} - {strings.startReg}</Text> {role === 'carpenter' && <TextInput placeholder={strings.nameOptional} value={name} onChangeText={setName} style={styles.input} />} <TextInput placeholder={strings.enterPhone} keyboardType='phone-pad' value={phone} onChangeText={setPhone} style={styles.input} /> <TouchableOpacity onPress={startRegister} style={styles.primaryBtn}><Text style={styles.primaryBtnText}>{strings.sendOtp}</Text></TouchableOpacity> <TouchableOpacity onPress={() => setScreen('home')} style={styles.linkBtn}><Text>{strings.back}</Text></TouchableOpacity> </View> );

const renderVerify = () => ( <View style={styles.card}> <Text style={styles.h2}>{strings.enterOtp} - {phone}</Text> <TextInput placeholder={strings.enterOtp} keyboardType='number-pad' value={otpInput} onChangeText={setOtpInput} style={styles.input} /> <TouchableOpacity onPress={verifyOtpAndSave} style={styles.primaryBtn}><Text style={styles.primaryBtnText}>{strings.verify}</Text></TouchableOpacity> <TouchableOpacity onPress={() => { setScreen('register'); setOtpSent(null); }} style={styles.linkBtn}><Text>{strings.back}</Text></TouchableOpacity> </View> );

const renderList = () => ( <View style={styles.card}> <Text style={styles.h2}>{strings.seeList}</Text> {carpenters.length === 0 ? <Text style={{marginTop:12}}>{strings.noCarpenters}</Text> : <FlatList data={carpenters} keyExtractor={(i)=>i.id} renderItem={({item}) => ( <View style={styles.row}> <View style={{flex:1}}> <Text style={{fontWeight:'700', fontSize:16}}>{item.name}</Text> <Text>{item.skills} • {item.city}</Text> </View> <TouchableOpacity onPress={() => callNumber(item.phone)} style={styles.callBtn}><Text style={{color:'#fff'}}>{strings.call}</Text></TouchableOpacity> </View> )} /> } <TouchableOpacity onPress={() => setScreen('home')} style={styles.linkBtn}><Text>{strings.back}</Text></TouchableOpacity> </View> );

return ( <SafeAreaView style={styles.container}> {screen === 'home' && renderHome()} {screen === 'register' && renderRegister()} {screen === 'verify' && renderVerify()} {screen === 'list' && renderList()} </SafeAreaView> ); }

const styles = StyleSheet.create({ container: {flex:1, backgroundColor:'#f3f4f6', padding:12}, card: {backgroundColor:'#fff', padding:16, borderRadius:12, marginTop:16, minHeight:200, elevation:2}, title: {fontSize:22, fontWeight:'800', textAlign:'center'}, subtitle: {textAlign:'center', marginTop:6, color:'#555'}, roleBtn: {padding:14, borderRadius:10, borderWidth:1, borderColor:'#eee', width:'48%', alignItems:'center'}, roleBtnActive: {backgroundColor:'#e6f9f0'}, roleText: {fontSize:16, fontWeight:'700'}, primaryBtn: {backgroundColor:'#1f7a68', padding:14, borderRadius:10, marginTop:12, alignItems:'center'}, primaryBtnText: {color:'#fff', fontWeight:'800', fontSize:16}, ghostBtn: {padding:12, marginTop:8, alignItems:'center'}, smallBtn: {padding:8, borderRadius:8, borderWidth:1, borderColor:'#ddd', backgroundColor:'#fff'}, input: {borderWidth:1, borderColor:'#eaeaea', borderRadius:10, padding:12, marginVertical:8, backgroundColor:'#fff'}, h2: {fontSize:18, fontWeight:'700', marginBottom:8}, linkBtn: {marginTop:10, alignItems:'center'}, row: {flexDirection:'row', alignItems:'center', paddingVertical:10, borderBottomWidth:1, borderColor:'#f0f0f0'}, callBtn: {backgroundColor:'#e64a19', paddingHorizontal:12, paddingVertical:8, borderRadius:8} });

/* ------------------ NOTES ------------------

This is an MVP single-file app (App.js). To use:

1. expo init myapp


2. npm install @react-native-async-storage/async-storage expo-speech


3. replace App.js with this file and run expo start



OTP here is a demo (shown in alert + spoken). For production, integrate Firebase Authentication (Phone Auth) or Twilio SMS via a backend.

To save carpenters across devices globally, integrate Firebase Firestore (replace AsyncStorage save/load with Firestore CRUD).
